type javaHybridRanker struct {
	Scores []rerank.RankResult
	Err    string
	Limit  int
}

func (m *javaHybridRanker) Rerank(context.Context, string, []string) ([]rerank.RankResult, error) {
	if m.Err != "" {
		return nil, errors.New(m.Err)
	}
	return append([]rerank.RankResult(nil), m.Scores...), nil
}
func (*javaHybridRanker) GetModelName() string         { return "oracle" }
func (*javaHybridRanker) GetModelID() string           { return "oracle" }
func (m *javaHybridRanker) MaxPassageRunes(string) int { return m.Limit }
func TestJavaHybridOracle(t *testing.T) {
	ctx := context.Background()
	fusion := []any{}
	hit := func(id string, score float64, match types.MatchType) *types.IndexWithScore {
		return &types.IndexWithScore{ID: "index-" + id, ChunkID: id, Content: fmt.Sprintf("content-%d", match), KnowledgeID: "doc", KnowledgeBaseID: "kb", SourceID: "source-" + id, Score: score, MatchType: match}
	}
	for _, config := range []*types.RetrievalConfig{nil, {RRFK: 1}, {RRFK: 60, RRFVectorWeight: 2, RRFKeywordWeight: 1}, {RRFK: 100, RRFVectorWeight: -1, RRFKeywordWeight: .9}} {
		for mode := 0; mode < 4; mode++ {
			for shift := 0; shift < 7; shift++ {
				v := [][]*types.IndexWithScore{{hit("a", .92, 0), hit("b", .73, 0), hit("a", .87, 0)}, {hit("c", .82, 0), hit("b", .75, 0)}}
				k := [][]*types.IndexWithScore{{hit("b", float64(15+shift), 1), hit("c", 2, 1), hit("d", 1, 1)}, {hit("a", .8, 1), hit("b", .7, 1)}}
				if mode == 0 {
					v = nil
				}
				if mode == 1 {
					k = nil
				}
				if mode == 3 {
					v = nil
					k = nil
				}
				input, _ := json.Marshal(map[string]any{"vectors": v, "keywords": k, "config": config})
				out := fuseOrDeduplicate(ctx, v, k, config)
				fusion = append(fusion, map[string]any{"input": json.RawMessage(input), "result": out})
			}
		}
	}
	passage := []any{}
	texts := []string{"# Heading\n\n**bold** and *italic*", "[![alt](img.png)](https://outside) link [display](url) ![x](img)", "```go\nfunc main(){}\n```", "$$x+y$$", "| a | b |\n|---|---|\n| c | d |", "<div>raw https://x/a more</div>", "> quoted\n1. numbered\n- bullet", "\u00a0 Chinese 中文 😀 \u3000", "", "![only](resource://one)", "# ![only](image)", "\n\n\nA\n\n\n\nB"}
	for _, text := range texts {
		for _, kind := range []string{"text", "faq", "image_ocr"} {
			for _, enrich := range []string{"", `[{"caption":"caption","ocr_text":"OCR"}]`, `[{"caption":null,"ocr_text":"OCR"}]`, `[{"caption":4,"caption":"fixed"}]`, `[null]`} {
				r := &types.SearchResult{ID: "row", Content: text, ChunkType: kind, KnowledgeTitle: "  document title ", ContextHeader: "Heading > Child", ImageInfo: enrich, ChunkMetadata: types.JSON(`{"generated_questions":[{"question":"first"},{"question":"second"}]}`)}
				raw, _ := json.Marshal(r)
				var row map[string]any
				json.Unmarshal(raw, &row)
				row["context_header"] = r.ContextHeader
				passage = append(passage, map[string]any{"row": row, "clean": reranking.CleanPassage(text), "enriched": reranking.EnrichedPassage(ctx, r), "model": reranking.ModelPassage(ctx, r)})
			}
		}
	}
	tokens := []any{}
	for _, text := range append(texts, []string{"南京市长江大桥", "中华人民共和国退款政策", "Echo open-weight models reduce cost", "他来到了网易杭研大厦", "中文😀混合ABC words words", "🧪 !! -- one two αβ İΣ"}...) {
		set := searchutil.TokenizeSimple(text)
		words := []string{}
		for w := range set {
			words = append(words, w)
		}
		sort.Strings(words)
		tokens = append(tokens, map[string]any{"text": text, "tokens": words})
	}
	rankCases := []any{}
	for _, threshold := range []float64{-.5, 0, .2, .3, .5, .8, 1.1} {
		for _, topk := range []int{0, 1, 2, 4} {
			for _, limit := range []int{0, 12} {
				for _, boost := range []float64{0, 2} {
					rows := []*types.SearchResult{{ID: "a", Content: "shared words echo", Score: .8, ChunkType: "text", KnowledgeTitle: "Title", Metadata: map[string]string{"existing": "one"}}, {ID: "b", Content: "shared words echo more", Score: .6, ChunkType: "faq"}, {ID: "c", Content: "南京市长江大桥 退款规则", Score: .75, ChunkType: "text", KnowledgeSource: "WEB_SEARCH"}, {ID: "d", Content: "graph evidence distinct", Score: 0, MatchType: types.MatchTypeGraph, ChunkType: "text"}, {ID: "empty", Content: "![empty](url)", Score: .99, KnowledgeTitle: "title only"}}
					scores := []rerank.RankResult{{Index: 1, RelevanceScore: .37}, {Index: 0, RelevanceScore: .55}, {Index: 2, RelevanceScore: .41}, {Index: 3, RelevanceScore: .22}, {Index: 99, RelevanceScore: .9}, {Index: -1, RelevanceScore: .8}}
					opts := reranking.Options{Threshold: threshold, TopK: topk, FallbackMinScore: .15, MaxCandidates: 0, FAQScoreBoost: boost}
					model := &javaHybridRanker{Scores: scores, Limit: limit}
					res := reranking.Rerank(ctx, model, "query", rows, opts)
					rankCases = append(rankCases, map[string]any{"rows": rows, "scores": scores, "limit": limit, "options": opts, "result": res})
				}
			}
		}
	}
	for _, scenario := range []string{"no_candidates", "model_error", "fallback_top1", "all_below", "duplicates", "max_candidates"} {
		rows := []*types.SearchResult{{ID: "a", Content: "first passage", Score: .8, ChunkType: "text"}, {ID: "b", Content: "second passage", Score: .9, ChunkType: "text"}, {ID: "c", Content: "graph passage", Score: 0, MatchType: types.MatchTypeGraph}}
		scores := []rerank.RankResult{{Index: 0, RelevanceScore: .1}, {Index: 1, RelevanceScore: .2}}
		opts := reranking.Options{Threshold: .8, TopK: 2, FallbackMinScore: .15}
		model := &javaHybridRanker{Scores: scores}
		switch scenario {
		case "no_candidates":
			rows = nil
		case "model_error":
			model.Err = "provider unavailable"
		case "all_below":
			model.Scores = []rerank.RankResult{{Index: 0, RelevanceScore: .01}}
		case "duplicates":
			model.Scores = []rerank.RankResult{{Index: 0, RelevanceScore: .9}, {Index: 0, RelevanceScore: .8}, {Index: 1, RelevanceScore: .82}}
		case "max_candidates":
			opts.MaxCandidates = 1
		}
		res := reranking.Rerank(ctx, model, "query", rows, opts)
		rankCases = append(rankCases, map[string]any{"rows": rows, "scores": model.Scores, "error": model.Err, "options": opts, "limit": 0, "result": res})
	}
	data := map[string]any{"fusion": fusion, "passage": passage, "tokens": tokens, "rerank": rankCases}
	f, err := os.Create(os.Getenv("WEKNORA_HYBRID_FIXTURE_OUTPUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	e := json.NewEncoder(f)
	e.SetIndent("", "  ")
	if err = e.Encode(data); err != nil {
		t.Fatal(err)
	}
}
