package service

import (
	"context"
	"encoding/json"
	rsqlite "github.com/Tencent/WeKnora/internal/application/repository/retriever/sqlite"
	"github.com/Tencent/WeKnora/internal/models/embedding"
	"github.com/Tencent/WeKnora/internal/models/rerank"
	"github.com/Tencent/WeKnora/internal/reranking"
	"github.com/Tencent/WeKnora/internal/types"
	"github.com/Tencent/WeKnora/internal/utils"
	vec "github.com/asg017/sqlite-vec-go-bindings/cgo"
	gsqlite "gorm.io/driver/sqlite"
	"gorm.io/gorm"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"testing"
)

func TestJavaHybridWireOracle(t *testing.T) {
	ctx := context.Background()
	os.Setenv("SSRF_WHITELIST", "127.0.0.1")
	utils.ResetSSRFWhitelistForTest()
	var mu sync.Mutex
	requests := []any{}
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		var body map[string]any
		if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
			panic(err)
		}
		mu.Lock()
		requests = append(requests, map[string]any{"path": r.URL.Path, "authorization": r.Header.Get("Authorization"), "body": body})
		mu.Unlock()
		w.Header().Set("Content-Type", "application/json")
		if strings.HasSuffix(r.URL.Path, "embeddings") {
			json.NewEncoder(w).Encode(map[string]any{"data": []any{map[string]any{"index": 0, "embedding": []float32{1, .125}}}})
			return
		}
		scores := []any{}
		docs := body["documents"].([]any)
		for i := len(docs) - 1; i >= 0; i-- {
			score := .63
			if strings.Contains(docs[i].(string), "beta") {
				score = .88
			} else if strings.Contains(docs[i].(string), "alpha") {
				score = .74
			}
			scores = append(scores, map[string]any{"index": i, "relevance_score": score})
		}
		json.NewEncoder(w).Encode(map[string]any{"results": scores})
	}))
	defer server.Close()
	vec.Auto()
	dir := t.TempDir()
	db, err := gorm.Open(gsqlite.Open(filepath.Join(dir, "schema.db")), &gorm.Config{})
	if err != nil {
		t.Fatal(err)
	}
	sd, err := db.DB()
	if err != nil {
		t.Fatal(err)
	}
	defer sd.Close()
	sd.SetMaxOpenConns(1)
	init, err := os.ReadFile("../../../migrations/sqlite/000000_init.up.sql")
	if err != nil {
		t.Fatal(err)
	}
	if err = db.Exec(string(init)).Error; err != nil {
		t.Fatal(err)
	}
	repo := rsqlite.NewSQLiteRetrieveEngineRepository(db)
	entries := []*types.IndexInfo{{SourceID: "alpha", ChunkID: "alpha", KnowledgeID: "doc-a", KnowledgeBaseID: "kb", TagID: "tag", Content: "alpha refund policy shared words", IsEnabled: true}, {SourceID: "beta", ChunkID: "beta", KnowledgeID: "doc-b", KnowledgeBaseID: "kb", TagID: "tag", Content: "beta refund policy other words additional", IsEnabled: true}, {SourceID: "china", ChunkID: "china", KnowledgeID: "doc-b", KnowledgeBaseID: "kb", Content: "中国退款政策 长江大桥", IsEnabled: true}, {SourceID: "foreign", ChunkID: "foreign", KnowledgeID: "foreign", KnowledgeBaseID: "foreign", Content: "beta refund policy foreign", IsEnabled: true}, {SourceID: "disabled", ChunkID: "disabled", KnowledgeID: "doc-a", KnowledgeBaseID: "kb", Content: "alpha refund policy disabled", IsEnabled: false}}
	vectors := map[string][]float32{"alpha": {1, 0}, "beta": {.8, .6}, "china": {.2, .98}, "foreign": {1, .1}, "disabled": {1, .125}}
	if err = repo.BatchSave(ctx, entries, map[string]any{"embedding": vectors}); err != nil {
		t.Fatal(err)
	}
	model := func(kind types.ModelType, id string) *types.Model {
		return &types.Model{ID: id, TenantID: 1, Name: "unknown", Source: types.ModelSourceRemote, Type: kind, Parameters: types.ModelParameters{Provider: "generic", BaseURL: server.URL + "/v1", APIKey: "synthetic-api-key"}}
	}
	embed, err := embedding.NewEmbedder(embedding.ConfigFromModel(model(types.ModelTypeEmbedding, "embed"), "", ""), nil, nil)
	if err != nil {
		t.Fatal(err)
	}
	ranker, err := rerank.NewReranker(rerank.ConfigFromModel(model(types.ModelTypeRerank, "rerank"), "", ""))
	if err != nil {
		t.Fatal(err)
	}
	cases := []any{}
	for mode := 0; mode < 3; mode++ {
		for scope := 0; scope < 4; scope++ {
			requests = []any{}
			query := "refund policy"
			if scope == 3 {
				query = "中国退款"
			}
			params := types.RetrieveParams{Query: query, KnowledgeBaseIDs: []string{"kb"}, TopK: 10}
			if scope == 1 {
				params.TagIDs = []string{"tag"}
			}
			if scope == 2 {
				params.KnowledgeIDs = []string{"doc-b"}
			}
			if mode != 2 {
				params.Embedding, err = embed.Embed(types.WithEmbedQuery(ctx), query)
				if err != nil {
					t.Fatal(err)
				}
			}
			if mode == 1 {
				params.RetrieverType = types.VectorRetrieverType
			} else if mode == 2 {
				params.RetrieverType = types.KeywordsRetrieverType
			}
			retrieved, e := repo.Retrieve(ctx, params)
			if e != nil {
				t.Fatal(e)
			}
			retrievedSnapshot, _ := json.Marshal(retrieved)
			vl := [][]*types.IndexWithScore{}
			kl := [][]*types.IndexWithScore{}
			for _, r := range retrieved {
				if r.RetrieverType == types.VectorRetrieverType {
					vl = append(vl, r.Results)
				} else {
					kl = append(kl, r.Results)
				}
			}
			fused := fuseOrDeduplicate(ctx, vl, kl, nil)
			rows := []*types.SearchResult{}
			for _, h := range fused {
				rows = append(rows, &types.SearchResult{ID: h.ChunkID, Content: h.Content, KnowledgeID: h.KnowledgeID, KnowledgeBaseID: h.KnowledgeBaseID, KnowledgeTitle: "Document " + h.KnowledgeID, Score: h.Score, MatchType: h.MatchType, ChunkType: "text"})
			}
			opts := reranking.Options{Threshold: .8, TopK: 2, FallbackMinScore: .15}
			out := reranking.Rerank(ctx, ranker, query, rows, opts)
			snapshot := append([]any(nil), requests...)
			cases = append(cases, map[string]any{"mode": mode, "params": params, "retrieved": json.RawMessage(retrievedSnapshot), "fused": fused, "rows": rows, "options": opts, "result": out, "requests": snapshot})
		}
	}
	f, e := os.Create(os.Getenv("WEKNORA_HYBRID_WIRE_FIXTURE_OUTPUT"))
	if e != nil {
		t.Fatal(e)
	}
	defer f.Close()
	enc := json.NewEncoder(f)
	enc.SetIndent("", "  ")
	if e = enc.Encode(map[string]any{"entries": entries, "vectors": vectors, "cases": cases}); e != nil {
		t.Fatal(e)
	}
}
