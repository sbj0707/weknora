package service

import (
	"context"
	"encoding/json"
	"github.com/Tencent/WeKnora/internal/reranking"
	"github.com/Tencent/WeKnora/internal/types"
	"os"
	"testing"
)

func TestJavaPassageMetadataOracle(t *testing.T) {
	images := []string{`[{"caption":"caption","attrs":{"source":{"schema":"s","prompt":"p","custom":true}}}]`, `[{"caption":"caption","attrs":{"source":"old-scalar"}}]`, `[{"caption":"caption","attrs":{"source":{"custom":4,"custom":true}}}]`, `[{"caption":"\ud800","ocr_text":"\udc00"}]`, `[{"caption":"caption","attrs":{"attrs":{"x":-0,"number":1e309}}}]`, `[{"caption":"caption","attrs":null}]`, `[{"caption":"first","caption":null}]`, `[{"caption":4,"caption":"last"}]`, `[null]`, `[]`}
	metadata := []string{`{"generated_questions":[null,{"question":"\ud800"}]}`, `{"generated_questions":[{"question":"first","question":null}]}`, `{"generated_questions":[{"question":4,"question":"last"}]}`, `{"generated_questions":[{"question":"yes","content_revision":1e9}]}`, `{"generated_questions":[{"question":"yes","content_revision":null}]}`, `null`, `{}`, `{"generated_questions":[{"QUESTION":"Unicode İΣ K ſ"}]}`}
	cases := []any{}
	for _, image := range images {
		for _, raw := range metadata {
			row := &types.SearchResult{ID: "row", Content: "body", ChunkType: "text", KnowledgeTitle: "Title", ContextHeader: "Header", ImageInfo: image, ChunkMetadata: types.JSON(raw)}
			cases = append(cases, map[string]any{"image": image, "metadata_raw": raw, "enriched": reranking.EnrichedPassage(context.Background(), row), "model": reranking.ModelPassage(context.Background(), row)})
		}
	}
	f, err := os.Create(os.Getenv("WEKNORA_PASSAGE_METADATA_FIXTURE_OUTPUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	e := json.NewEncoder(f)
	e.SetIndent("", "  ")
	if err = e.Encode(cases); err != nil {
		t.Fatal(err)
	}
}
